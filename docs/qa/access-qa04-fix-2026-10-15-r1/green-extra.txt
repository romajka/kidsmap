async function extraLayout(p,name,expectedHeadings=null){const d=await p.evaluate(()=>{
 const panel=document.querySelector('.owner-team'),pr=panel.getBoundingClientRect();const within=(r,c)=>r.x>=c.x-.5&&r.right<=Math.min(c.right,pr.right,innerWidth)+.5;
 return{headings:[...panel.querySelectorAll('.owner-team-item h3')].map(e=>e.textContent.trim()),items:[...panel.querySelectorAll('.owner-team-item')].map(c=>{const cr=c.getBoundingClientRect();return{metadata:[...c.querySelectorAll('.meta')].map(e=>{const range=document.createRange();range.selectNodeContents(e);const rr=[...range.getClientRects()].filter(r=>r.width>.5&&r.height>.5);return{text:e.textContent.trim(),scroll:e.scrollWidth,client:e.clientWidth,allTextFits:rr.every(r=>within(r,cr))};}),forms:[...c.querySelectorAll('form')].map(e=>{const r=e.getBoundingClientRect();return{x:r.x,right:r.right,fits:within(r,cr)};})};})};});
record(name+': metadata text and forms fit',d.items.every(c=>c.metadata.every(e=>e.allTextFits&&e.scroll<=e.client+.5)&&c.forms.every(e=>e.fits)),d);
if(expectedHeadings)record(name+': complete original headings retained',JSON.stringify(d.headings.slice().sort())===JSON.stringify(expectedHeadings.slice().sort()),{actual:d.headings,expected:expectedHeadings});}
async function keyboardCheck(p){const select=p.locator('.owner-team-role-form select'),value=await select.inputValue();await p.mouse.move(0,0);await select.focus();await p.keyboard.press('Space');await p.keyboard.press('ArrowUp');await p.keyboard.press('Escape');await select.selectOption(value);await p.keyboard.press('Space');await p.keyboard.press('Enter');await p.keyboard.press('Escape');await p.keyboard.press('Tab');const btn=p.locator('.owner-team-role-form button');const d=await btn.evaluate(e=>{const s=getComputedStyle(e),r=e.getBoundingClientRect();return{active:document.activeElement===e,visible:e.matches(':focus-visible'),outlineStyle:s.outlineStyle,outlineWidth:s.outlineWidth,shadow:s.boxShadow,x:r.x,right:r.right,width:innerWidth};});record('native select and role button keyboard focus',d.active&&d.visible&&((d.outlineStyle!=='none'&&parseFloat(d.outlineWidth)>0)||d.shadow!=='none'),d);record('focused role button fits viewport',d.x>=0&&d.right<=d.width+.5,d);await p.keyboard.press('Shift+Tab');record('ShiftTab returns to native role select',await select.evaluate(e=>document.activeElement===e));await p.keyboard.press('Tab');}
